package envoy.authz

import rego.v1

# =============================================================================
# OPA Policy — Zero Trust Authorization for ZTALeaks Nuclear Plant
# =============================================================================
# Access decision based on 4 independent dimensions, all required:
#
#   1. Public path     — some routes are open (login, register, JWKS, etc.)
#   2. Tier admission  — certificate and/or TPM determine the tier (0/1/2);
#                        each route declares its required `min_tier`
#   3. Role            — each (path, method) has its allowed `roles` list
#   4. Clearance       — PUBLIC<INTERNAL<CONFIDENTIAL<SECRET<TOP_SECRET hierarchy;
#                        the user must have ≥ the route's `min_clearance`
#
# Expected input (populated by the security-orchestrator):
#   {
#     "request": {"method", "path", "headers"},
#     "claims":  {"sub", "role", "clearance_level", "mfa_verified", "device_id"} | null,
#     "cert_present": bool,
#     "cert_subject": str (opt),
#     "tpm_verified": bool,
#     "zone_id": str (opt),
#     "ai":      {"score": 0..0.99, "confidence": "high"|"low"} (opt),
#     "context": {"hour_of_day": 0..23, "day_of_week": 0..6,
#                  "session_age_seconds": int, "client_ip": str} (opt)
#   }
#
# Exposed outputs:
#   - allow:        bool — final decision (the PEP consumes it)
#   - risk_bucket:  "low"|"medium"|"high" — used by the orchestrator for audit
#   - risk_score:   int — deterministic fallback (only for confidence=low)
# =============================================================================

default allow := false

# -----------------------------------------------------------------------------
# 1. PUBLIC PATHS — open access, no JWT required
# -----------------------------------------------------------------------------
public_paths := {
    "/",
    "/health",
    "/login",
    "/register",
    "/materials",
    "/api/v1/auth/login",
    "/api/v1/auth/register",
    "/api/v1/auth/verify-otp",
    # /api/v1/auth/register/begin now requires authentication: iam-service
    # reads the X-Current-User header injected by the security-orchestrator after
    # JWT verification. Explicit match below in route_rules.
    "/api/v1/auth/login/begin",
    "/api/v1/auth/login/finish",
    "/.well-known/jwks.json",
    "/reserved",
}

allow if {
    input.request.path in public_paths
}

allow if {
    startswith(input.request.path, "/static/")
}

# -----------------------------------------------------------------------------
# 2. TIER ADMISSION
#    0 = neither cert nor TPM (login with password+OTP only)
#    1 = client certificate only (mTLS recognised)
#    2 = certificate + enrolled TPM/WebAuthn
# -----------------------------------------------------------------------------
user_tier := 2 if {
    input.cert_present
    input.tpm_verified
}

user_tier := 1 if {
    input.cert_present
    not input.tpm_verified
}

user_tier := 0 if {
    not input.cert_present
}

# -----------------------------------------------------------------------------
# 3. CLEARANCE HIERARCHY
# -----------------------------------------------------------------------------
clearance_order := {
    "PUBLIC": 0,
    "INTERNAL": 1,
    "CONFIDENTIAL": 2,
    "SECRET": 3,
    "TOP_SECRET": 4,
}

# -----------------------------------------------------------------------------
# 4. ROLE × ROUTE MATRIX
#    For each (path, method): {roles, min_tier, min_clearance}.
#    Path matching: exact, or "{path}/" prefix (e.g. /personnel/EMP-001).
# -----------------------------------------------------------------------------
route_rules := {
    # TPM enrollment: the user must already be authenticated (post password+OTP)
    # to register a device. Tier 1 (mTLS) required because the enrollment is blind.
    "/api/v1/auth/register/begin": {
        "POST": {"roles": {"plant_manager", "operator", "maintenance_technician", "radiation_protection_officer", "security_officer", "inspector"}, "min_tier": 1, "min_clearance": "PUBLIC"},
    },
    "/api/v1/auth/register/finish": {
        "POST": {"roles": {"plant_manager", "operator", "maintenance_technician", "radiation_protection_officer", "security_officer", "inspector"}, "min_tier": 1, "min_clearance": "PUBLIC"},
    },
    "/api/v1/personnel": {
        "GET":    {"roles": {"security_officer", "plant_manager", "inspector"}, "min_tier": 1, "min_clearance": "INTERNAL"},
        "POST":   {"roles": {"plant_manager"}, "min_tier": 2, "min_clearance": "SECRET"},
        "PUT":    {"roles": {"plant_manager"}, "min_tier": 2, "min_clearance": "SECRET"},
        "DELETE": {"roles": {"plant_manager"}, "min_tier": 2, "min_clearance": "SECRET"},
    },
    "/api/v1/zones": {
        "GET":    {"roles": {"operator", "plant_manager", "inspector", "maintenance_technician", "radiation_protection_officer", "security_officer"}, "min_tier": 0, "min_clearance": "PUBLIC"},
        "POST":   {"roles": {"plant_manager"}, "min_tier": 2, "min_clearance": "SECRET"},
        "PUT":    {"roles": {"plant_manager"}, "min_tier": 2, "min_clearance": "SECRET"},
        "DELETE": {"roles": {"plant_manager"}, "min_tier": 2, "min_clearance": "SECRET"},
    },
    "/api/v1/badges": {
        "GET":    {"roles": {"security_officer", "plant_manager", "inspector"}, "min_tier": 1, "min_clearance": "INTERNAL"},
        "POST":   {"roles": {"plant_manager"}, "min_tier": 2, "min_clearance": "CONFIDENTIAL"},
        "PUT":    {"roles": {"plant_manager"}, "min_tier": 2, "min_clearance": "CONFIDENTIAL"},
        "DELETE": {"roles": {"plant_manager"}, "min_tier": 2, "min_clearance": "CONFIDENTIAL"},
    },
    "/api/v1/reactor-parameters": {
        "GET":    {"roles": {"operator", "plant_manager", "inspector"}, "min_tier": 1, "min_clearance": "CONFIDENTIAL"},
        "POST":   {"roles": {"plant_manager"}, "min_tier": 2, "min_clearance": "SECRET"},
        "PUT":    {"roles": {"plant_manager"}, "min_tier": 2, "min_clearance": "SECRET"},
        "DELETE": {"roles": {"plant_manager"}, "min_tier": 2, "min_clearance": "SECRET"},
    },
    "/api/v1/maintenance-orders": {
        "GET":    {"roles": {"maintenance_technician", "plant_manager"}, "min_tier": 1, "min_clearance": "INTERNAL"},
        "POST":   {"roles": {"maintenance_technician", "plant_manager"}, "min_tier": 1, "min_clearance": "INTERNAL"},
        "PUT":    {"roles": {"maintenance_technician", "plant_manager"}, "min_tier": 1, "min_clearance": "INTERNAL"},
        "DELETE": {"roles": {"maintenance_technician", "plant_manager"}, "min_tier": 2, "min_clearance": "CONFIDENTIAL"},
    },
    "/api/v1/documents": {
        "GET":    {"roles": {"operator", "plant_manager", "inspector", "maintenance_technician", "radiation_protection_officer", "security_officer"}, "min_tier": 0, "min_clearance": "PUBLIC"},
        "POST":   {"roles": {"plant_manager"}, "min_tier": 2, "min_clearance": "CONFIDENTIAL"},
        "PUT":    {"roles": {"plant_manager"}, "min_tier": 2, "min_clearance": "CONFIDENTIAL"},
        "DELETE": {"roles": {"plant_manager"}, "min_tier": 2, "min_clearance": "CONFIDENTIAL"},
    },
    "/api/v1/nuclear-materials": {
        "GET":    {"roles": {"plant_manager", "inspector", "radiation_protection_officer"}, "min_tier": 2, "min_clearance": "SECRET"},
        "POST":   {"roles": {"plant_manager"}, "min_tier": 2, "min_clearance": "TOP_SECRET"},
        "PUT":    {"roles": {"plant_manager"}, "min_tier": 2, "min_clearance": "TOP_SECRET"},
        "DELETE": {"roles": {"plant_manager"}, "min_tier": 2, "min_clearance": "TOP_SECRET"},
    },
}

# matched_route finds the route_rules entry matching input.request.path:
# exact match OR the "{key}/" prefix (allows /{id} subpaths).
matched_route := key if {
    some key, _ in route_rules
    input.request.path == key
}

matched_route := key if {
    some key, _ in route_rules
    startswith(input.request.path, concat("", [key, "/"]))
}

# -----------------------------------------------------------------------------
# 5. Main ALLOW: requires sufficient claims, tier, role and clearance.
#    The `risk_bucket != "high"` guard lets the anomaly-detection override
#    (section 6) apply even when the user would normally be authorised.
# -----------------------------------------------------------------------------
allow if {
    authn_authz_ok
    risk_bucket != "high"
}

authn_authz_ok if {
    # Claims present (authenticated user)
    input.claims.sub

    # Route match in the matrix
    rule := route_rules[matched_route][input.request.method]

    # Tier admission
    user_tier >= rule.min_tier

    # Role
    input.claims.role in rule.roles

    # Clearance
    clearance_order[input.claims.clearance_level] >= clearance_order[rule.min_clearance]
}

# -----------------------------------------------------------------------------
# 6. AI RISK SCORING — consumes the score from the AI microservice or computes a fallback.
#
# Confidence "high" (the AI answered) → direct bands on the score.
# Confidence "low" (AI down/timeout) → deterministic score from the context.
# `ai` input absent → bucket "low" (backward-compatible with the historical tests).
#
# Bands:
#   "low":    score < 0.3              → no impact
#   "medium": 0.3 ≤ score < 0.7        → allow + audit log (orchestrator)
#   "high":   score ≥ 0.7              → deny override
# -----------------------------------------------------------------------------
default risk_bucket := "low"

risk_bucket := "low" if {
    input.ai.confidence == "high"
    input.ai.score < 0.3
}

risk_bucket := "medium" if {
    input.ai.confidence == "high"
    input.ai.score >= 0.3
    input.ai.score < 0.7
}

risk_bucket := "high" if {
    input.ai.confidence == "high"
    input.ai.score >= 0.7
}

# Fallback: the AI failed → compute the bucket from the available context.
risk_bucket := bucket if {
    input.ai.confidence == "low"
    bucket := fallback_bucket
}

# Fallback score: sum of deterministic indicators (0..100).
risk_score := s if {
    s := off_hours_score + missing_cert_score + stale_session_score
}

# Off-hours: outside 06-22 and non-operational role (operators work shifts).
default off_hours_score := 0

off_hours_score := 25 if {
    input.context.hour_of_day < 6
    input.claims.role != "operator"
}

off_hours_score := 25 if {
    input.context.hour_of_day >= 22
    input.claims.role != "operator"
}

# Missing certificate on a route that requires tier ≥ 1.
default missing_cert_score := 0

missing_cert_score := 20 if {
    not input.cert_present
    rule := route_rules[matched_route][input.request.method]
    rule.min_tier >= 1
}

# Stale session: > 8h since login.
default stale_session_score := 0

stale_session_score := 15 if {
    input.context.session_age_seconds > 28800
}

# Score → bucket mapping for the fallback.
fallback_bucket := "high" if risk_score >= 50

fallback_bucket := "medium" if {
    risk_score >= 25
    risk_score < 50
}

fallback_bucket := "low" if risk_score < 25
